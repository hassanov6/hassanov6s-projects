📥 Export des données pour synchronisation APK

⚡ Option 1 - Directement depuis ce site :

  1. Cliquez sur le bouton "📤 Exporter les sessions complétées"
  2. Le JSON est copié dans le presse-papier
  3. Dans l'APK, cliquez sur "Importer depuis le site"
  4. Collez le JSON dans le champ de texte

📋 Option 2 - Depuis site1-omega-five.vercel.app :

  1. Allez sur site1-omega-five.vercel.app dans un navigateur PC
  2. Appuyez sur F12 pour ouvrir la console
  3. Copiez et collez ce script :
    4. (function() {
    const exportData = {
        export_date: new Date().toISOString(),
        version: "1.0",
        completed_sessions: [],
        total_completed: 0
    };

    // Récupérer depuis window
    if (window.userProgress && Object.keys(window.userProgress).length > 0) {
        for (const date in window.userProgress) {
            const progress = window.userProgress[date];
            const session = window.trainingProgram?.sessions?.[date];

            if (progress && progress.completed) {
                const zoneTimes = {
                    z1: progress.zoneTimes?.[0] || 0,
                    z2: progress.zoneTimes?.[1] || 0,
                    z3: progress.zoneTimes?.[2] || 0,
                    z4: progress.zoneTimes?.[3] || 0,
                    z5: progress.zoneTimes?.[4] || 0
                };

                exportData.completed_sessions.push({
                    date: date,
                    session_type: session?.session_type || "Endurance",
                    duration_minutes: session?.duration_minutes || 0,
                    description: session?.description || "",
                    target_zone: session?.target_zone || "Z1",
                    completed: true,
                    completed_at: progress.completedAt || new Date().toISOString(),
                    zone_times: zoneTimes,
                    notes: progress.notes || "",
                    rating: progress.rating || null
                });
            }
        }
    }

    exportData.total_completed = exportData.completed_sessions.length;

    console.log("=== COPIEZ CE JSON ===");
    console.log(JSON.stringify(exportData, null, 2));
    console.log("=== FIN DU JSON ===");

    // Copier dans le presse-papier
    navigator.clipboard.writeText(JSON.stringify(exportData, null, 2))
        .then(() => alert("✅ " + exportData.total_completed + " sessions exportées !\nLe JSON est copié dans le presse-papier."))
        .catch(err => console.error("Erreur:", err));
})();
  4. Appuyez sur Entrée
  5. Le JSON est copié dans le presse-papier
  6. Importez dans l'APK comme dans l'Option 1
